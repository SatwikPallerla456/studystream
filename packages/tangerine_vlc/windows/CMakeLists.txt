cmake_minimum_required(VERSION 3.14)
project(tangerine_vlc LANGUAGES CXX)
set(PLUGIN_NAME tangerine_vlc_plugin)
set(VLC_ROOT "$ENV{VLC_ROOT}" CACHE PATH "VideoLAN VLC 3 runtime with SDK (matching target architecture)")
find_path(VLC_INCLUDE_DIR vlc/vlc.h PATHS "${VLC_ROOT}/sdk/include" "${VLC_ROOT}/include" REQUIRED)
add_library(${PLUGIN_NAME} SHARED tangerine_vlc_plugin.cpp)
apply_standard_settings(${PLUGIN_NAME})
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL NOMINMAX)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(${PLUGIN_NAME} PRIVATE "${VLC_INCLUDE_DIR}")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter flutter_wrapper_plugin advapi32)
file(GLOB VLC_DLLS "${VLC_ROOT}/*.dll")
if(NOT VLC_DLLS OR NOT EXISTS "${VLC_ROOT}/plugins")
  message(FATAL_ERROR "VLC_ROOT must contain VLC 3 runtime DLLs and plugins as well as the SDK")
endif()
set(tangerine_vlc_bundled_libraries ${VLC_DLLS} PARENT_SCOPE)
install(DIRECTORY "${VLC_ROOT}/plugins" DESTINATION "." COMPONENT Runtime)
foreach(license COPYING.txt COPYING.LIB AUTHORS.txt)
  if(EXISTS "${VLC_ROOT}/${license}")
    install(FILES "${VLC_ROOT}/${license}" DESTINATION "licenses/VideoLAN" COMPONENT Runtime)
  endif()
endforeach()
