cmake_minimum_required(VERSION 3.13)
project(tangerine_vlc LANGUAGES CXX)
set(PLUGIN_NAME tangerine_vlc_plugin)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBVLC REQUIRED IMPORTED_TARGET libvlc>=3.0 libvlc<4.0)
add_library(${PLUGIN_NAME} SHARED tangerine_vlc_plugin.cc)
apply_standard_settings(${PLUGIN_NAME})
target_compile_features(${PLUGIN_NAME} PRIVATE cxx_std_17)
target_compile_options(${PLUGIN_NAME} PRIVATE -fexceptions)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter PkgConfig::LIBVLC)
set(tangerine_vlc_bundled_libraries "" PARENT_SCOPE)
